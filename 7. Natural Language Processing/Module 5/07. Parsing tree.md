# 💻 Step 7 — Code (NLTK Tree)

```python
# 👉 NLTK import kar rahe hain
import nltk

# 👉 CFG = Context Free Grammar (rules define karne ke liye)
from nltk import CFG

# 👉 ChartParser = grammar ke basis par sentence parse karta hai
from nltk.parse import ChartParser

# 👉 Grammar rules define kar rahe hain
grammar = CFG.fromstring("""
# S = Sentence → Noun Phrase + Verb Phrase
S -> NP VP

# NP = Noun Phrase → Ram ya Sita
NP -> 'Ram' | 'Sita'

# VP = Verb Phrase → Verb + Noun Phrase
VP -> V NP

# V = Verb → loves
V -> 'loves'
""")

# 👉 Parser object bana rahe hain using grammar
parser = ChartParser(grammar)

# 👉 Sentence ko words me tod rahe hain
sentence = "Ram loves Sita".split()

# 👉 Parser se sentence ka tree nikal rahe hain
for tree in parser.parse(sentence):
    # 👉 Tree print karega (structure dikhayega)
    print(tree)
```

---

## Output:

```
(S (NP Ram) (VP (V loves) (NP Sita)))
```
